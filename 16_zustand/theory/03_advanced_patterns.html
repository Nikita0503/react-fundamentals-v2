Selectors — Why They Matter for Performance

// ❌ Bad — component re-renders on ANY store change
const Component = () => {
  const store = useTodoStore(); // subscribes to entire store
  return <p>{store.todos.length}</p>;
};

// ✅ Good — component only re-renders when todos.length changes
const Component = () => {
  const todoCount = useTodoStore((state) => state.todos.length);
  return <p>{todoCount}</p>;
};
  • The function you pass to the hook IS the selector
  • Zustand uses Object.is to compare previous and next selected value
  • If the selected value didn't change → no re-render
  • Rule: always select the minimum data your component needs

Selecting Multiple Values

// Option 1: multiple selectors (each triggers re-render independently)
const todos = useTodoStore((state) => state.todos);
const addTodo = useTodoStore((state) => state.addTodo);

// Option 2: shallow comparison for objects (use when selecting multiple fields)
import { useShallow } from "zustand/react/shallow";

const { todos, addTodo } = useTodoStore(
  useShallow((state) => ({ todos: state.todos, addTodo: state.addTodo }))
);

useShallow does shallow comparison on the returned object — prevents re-render if individual fields didn't change.

Persist Middleware — Save State to localStorage

import { create } from "zustand";
import { persist } from "zustand/middleware";

type ThemeStore = {
  theme: "light" | "dark";
  toggleTheme: () => void;
};

const useThemeStore = create<ThemeStore>()(
  persist(
    (set) => ({
      theme: "light",
      toggleTheme: () =>
        set((state) => ({ theme: state.theme === "light" ? "dark" : "light" })),
    }),
    { name: "theme-storage" } // key in localStorage
  )
);
  • State automatically saved to localStorage on every change
  • On page reload, state is restored from localStorage
  • name is the localStorage key

Immer Middleware — Mutable Syntax

import { create } from "zustand";
import { immer } from "zustand/middleware/immer";

type Todo = { id: number; text: string; done: boolean };

type TodoStore = {
  todos: Todo[];
  toggleTodo: (id: number) => void;
};

const useTodoStore = create<TodoStore>()(
  immer((set) => ({
    todos: [],
    toggleTodo: (id) =>
      set((state) => {
        const todo = state.todos.find((t) => t.id === id);
        if (todo) todo.done = !todo.done; // direct mutation — Immer handles immutability
      }),
  }))
);
  • Install: npm install immer
  • Same concept as in Redux Toolkit — write mutations, Immer produces immutable updates

Async Actions — Fetching Data

import { useEffect } from "react";
import { create } from "zustand";
import { useShallow } from "zustand/react/shallow";

type User = { id: string; name: string; email: string };

type UserStore = {
  user: User | null;
  loading: boolean;
  error: string | null;
  fetchUser: (id: string) => Promise<void>;
};

const useUserStore = create<UserStore>((set) => ({
  user: null,
  loading: false,
  error: null,

  fetchUser: async (id) => {
    set({ loading: true, error: null });
    try {
      const res = await fetch(`/api/users/${id}`);
      const user = await res.json();
      set({ user, loading: false });
    } catch (err) {
      set({ error: "Failed to fetch user", loading: false });
    }
  },
}));

// Usage:
const Profile = ({ userId }: { userId: string }) => {
  const { user, loading, fetchUser } = useUserStore(
    useShallow((s) => ({ user: s.user, loading: s.loading, fetchUser: s.fetchUser }))
  );

  useEffect(() => {
    fetchUser(userId);
  }, [userId, fetchUser]);

  if (loading) return <p>Loading...</p>;
  return <p>{user?.name}</p>;
};
  • Actions can be async — just use async/await and call set when ready
  • Note: for server state with caching/refetching, TanStack Query is better. Zustand async is for simple cases or client-side async logic.

DevTools Middleware

import { create } from "zustand";
import { devtools } from "zustand/middleware";

type MyStore = {
  count: number;
  increment: () => void;
};

const useStore = create<MyStore>()(
  devtools(
    (set) => ({
      count: 0,
      increment: () => set((state) => ({ count: state.count + 1 })),
    }),
    { name: "MyStore" }
  )
);
  • Works with Redux DevTools browser extension
  • See state changes, time-travel debug

Combining Middleware

import { create } from "zustand";
import { devtools, persist } from "zustand/middleware";
import { immer } from "zustand/middleware/immer";

type MyStore = {
  count: number;
  increment: () => void;
};

const useStore = create<MyStore>()(
  devtools(
    persist(
      immer((set) => ({
        count: 0,
        increment: () =>
          set((state) => {
            state.count += 1;
          }),
      })),
      { name: "my-storage" }
    ),
    { name: "MyStore" }
  )
);

Middleware composes from inside out: immer → persist → devtools
