Який з наступних компонентів є простим, а який розумним?


Компонент A:

import React, { useState } from 'react';

const ComponentA = () => {
  const [count, setCount] = useState(0);

  const handleClick = () => {
    setCount(count + 1);
  }

  return (
    <div>
      <div>Count: {count}</div>
      <button onClick={handleClick}>Increment</button>
    </div>
  );
};

export default ComponentA;
