useMutation — For Creating, Updating, or Deleting Data

import { useMutation, useQueryClient } from "@tanstack/react-query";

type CreatePostDto = { title: string; body: string };

const CreatePostForm = () => {
  const queryClient = useQueryClient();

  const { mutate, isPending, isError, error } = useMutation({
    mutationFn: (newPost: CreatePostDto) =>
      fetch("/api/posts", {
        method: "POST",
        headers: { "Content-Type": "application/json" },
        body: JSON.stringify(newPost),
      }).then(res => res.json()),

    onSuccess: () => {
      // Invalidate the posts cache — triggers automatic refetch
      queryClient.invalidateQueries({ queryKey: ["posts"] });
    },
  });

  const handleSubmit = (data: CreatePostDto) => {
    mutate(data);
  };

  return (
    <form onSubmit={...}>
      {/* form fields */}
      <button type="submit" disabled={isPending}>
        {isPending ? "Creating..." : "Create Post"}
      </button>
      {isError && <p>Error: {error.message}</p>}
    </form>
  );
};

Key Difference from useQuery

  • useQuery runs automatically (on mount, on key change)
  • useMutation runs manually (when you call mutate())

Cache Invalidation — The Key Concept

  • After a mutation succeeds, the related cached data is stale
  • queryClient.invalidateQueries({ queryKey: ["posts"] }) marks all queries with that key as stale
  • Stale queries automatically refetch → UI updates with fresh data
  • This is how "add post → list updates automatically" works

Callbacks

useMutation({
  mutationFn: ...,
  onSuccess: (data, variables) => {
    // Runs after successful mutation
    // data = response from server
    // variables = what you passed to mutate()
  },
  onError: (error) => {
    // Runs if mutation fails
  },
  onSettled: () => {
    // Runs after success OR error (like finally)
  },
});

Full CRUD Example — Posts List with Add and Delete

const PostsPage = () => {
  const queryClient = useQueryClient();

  // READ
  const { data: posts, isLoading } = useQuery<Post[]>({
    queryKey: ["posts"],
    queryFn: () => fetch("/api/posts").then(r => r.json()),
  });

  // CREATE
  const addMutation = useMutation({
    mutationFn: (newPost: CreatePostDto) =>
      fetch("/api/posts", { method: "POST", headers: { "Content-Type": "application/json" }, body: JSON.stringify(newPost) }).then(r => r.json()),
    onSuccess: () => queryClient.invalidateQueries({ queryKey: ["posts"] }),
  });

  // DELETE
  const deleteMutation = useMutation({
    mutationFn: (id: number) =>
      fetch(`/api/posts/${id}`, { method: "DELETE" }),
    onSuccess: () => queryClient.invalidateQueries({ queryKey: ["posts"] }),
  });

  if (isLoading) return <p>Loading...</p>;

  return (
    <div>
      <button onClick={() => addMutation.mutate({ title: "New Post", body: "Content" })}>
        Add Post
      </button>
      <ul>
        {posts?.map(post => (
          <li key={post.id}>
            {post.title}
            <button onClick={() => deleteMutation.mutate(post.id)}>Delete</button>
          </li>
        ))}
      </ul>
    </div>
  );
};

Add or delete a post → cache invalidates → list refetches → UI updates. No manual state management.

Optimistic Updates (Brief Mention)

  • Instead of waiting for server response, update the UI immediately
  • If the server fails, roll back to previous state
  • TanStack Query has built-in support via onMutate + onError rollback
  • Advanced pattern — useful for real-time feel (likes, toggles, reordering)
