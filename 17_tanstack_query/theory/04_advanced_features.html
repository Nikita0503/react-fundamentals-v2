Pagination

const PostList = ({ page }: { page: number }) => {
  const { data, isLoading, isPlaceholderData } = useQuery({
    queryKey: ["posts", { page }],
    queryFn: () => fetch(`/api/posts?page=${page}`).then(r => r.json()),
    placeholderData: (previousData) => previousData, // keep showing old data while new page loads
  });

  return (
    <div>
      {isLoading ? <p>Loading...</p> : (
        <ul style={{ opacity: isPlaceholderData ? 0.5 : 1 }}>
          {data?.map((post: Post) => <li key={post.id}>{post.title}</li>)}
        </ul>
      )}
    </div>
  );
};
  • placeholderData: (prev) => prev — shows previous page data while new page loads (no loading flash)
  • isPlaceholderData — true when showing placeholder (useful for dimming UI)

Infinite Scroll with useInfiniteQuery

import { useInfiniteQuery } from "@tanstack/react-query";

type PostsResponse = { posts: Post[]; nextCursor: number | null };

const InfinitePostList = () => {
  const { data, fetchNextPage, hasNextPage, isFetchingNextPage, isLoading } = useInfiniteQuery<PostsResponse>({
    queryKey: ["posts", "infinite"],
    queryFn: ({ pageParam }) =>
      fetch(`/api/posts?cursor=${pageParam}`).then(r => r.json()),
    initialPageParam: 0,
    getNextPageParam: (lastPage) => lastPage.nextCursor,
  });

  if (isLoading) return <p>Loading...</p>;

  return (
    <div>
      {data?.pages.map((page, i) => (
        <div key={i}>
          {page.posts.map(post => <p key={post.id}>{post.title}</p>)}
        </div>
      ))}
      <button onClick={() => fetchNextPage()} disabled={!hasNextPage || isFetchingNextPage}>
        {isFetchingNextPage ? "Loading more..." : hasNextPage ? "Load More" : "No more posts"}
      </button>
    </div>
  );
};
  • getNextPageParam — tells TQ how to get the next page (cursor, offset, page number)
  • data.pages — array of all loaded pages
  • fetchNextPage() — loads the next page
  • hasNextPage — false when getNextPageParam returns null/undefined

Select — Transform Data

const { data: postTitles } = useQuery({
  queryKey: ["posts"],
  queryFn: fetchPosts,

  • select transforms cached data before returning to component
  • Component only re-renders when the selected (transformed) value changes
