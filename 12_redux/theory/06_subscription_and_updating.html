Subscribe (оновлення даних)

Повертаємося до файлу UserData:

/** /components/UserData.js */

const UserData = () => {
  const [name, setName] = React.useState('');
  const [surname, setSurname] = React.useState('');

  return (
    <div>
      <input value={name} onChange={(event) => setName(event.target.value)} />
      <input value={surname} onChange={(event) => setSurname(event.target.value)} />
    </div>
  );
}

export default UserData;

name та surname зараз знаходяться в середині Store, тож реактивні змінні в середині компоненту більше не потрібні.

useSelector

Для того щоб отримати доступ до name, surname у View/UI (компоненті) можна використовувати хук useSelector з пакету 'react-redux':

/** /components/UserData.js */

import { useSelector } from 'react-redux';

const UserData = () => {
  const name = useSelector(state => state.user.name);
  const surname = useSelector(state => state.user.surname);

  return (
    <div>
      <input value={name} />
      <input value={surname} />
    </div>
  );
}

export default UserData;

useSelector повертає значення, котрі знаходяться за адресою, котра будується по принципу - state.REDUCER_NAME.FIELD_NAME (state.НАЗВА_REDUCER.НАЗВА_ПОЛЯ).

Так як у RootReducer для userReducer ми надали ключ user, а name знаходиться в середині userReducer, маємо адресу - state.user.name.

useDispatch

Зараз View/UI відображає дані зі Store. Але при редагуванні input, name та surname досі змінюватися не будуть.

Тому Потрібно навчити компонент оновлювати name та surname. Для цього використовується функція Dispatch.

Хук useDispatch з пакету 'react-redux' повертає функцію Dispatch:

/** /components/UserData.js */

import { useSelector, useDispatch } from 'react-redux';
import { setName, setSurname } from '../redux/actions';

const UserData = () => {
  const dispatch = useDispatch();

  const name = useSelector(state => state.user.name);
  const surname = useSelector(state => state.login.surname);

  return (
    <div>
      <input value={name} onChange={(event) => dispatch(setName(event.target.value))} />
      <input value={surname} onChange={(event) => dispatch(setSurname(event.target.value))} />
    </div>
  );
}

export default UserData;

Dispatch приймає Action (який формує та повертає Action Creator), та направляє його у Reducer, де відбувається оновлення Store.

Увага:

  • Не намагайтесь оновити дані у Store без використання Dispatch.
  • useSelector, useDispatch не є єдиними шляхами до взаємодії зі Store. Існує спосіб взаємодії через функцію connect.
