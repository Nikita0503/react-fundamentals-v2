Підключення Redux до проекту

Зараз App.js файл виглядає так:

/** App.js */

const App = () => {
  return (
    <UserData/>
  );
};

export default App;

Щоб дати усім компонентам доступ до Store, необхідно їх огорнути у Provider з пакету 'react-redux' та надати йому Store:

/** App.js */

import { Provider } from 'react-redux';
import { store } from './redux/store';
import UserData from './UserData';

const App = () => {

  return (
    <Provider store={store}> 
      <UserData/>
    </Provider>
  );
}


export default App;

Тепер кожен компонент, котрий використовується в середині компонента App має змогу отримати та оновити дані Store.
