2. What is wrong with this code?

import { useSelector } from "react-redux";

const UserName = () => {
  const name = useSelector((state) => state.user.name);
  return <span>{name}</span>;
};
You can't use useSelector inside a functional component
(x) The state parameter has no type annotation — it will be typed as unknown, losing type safety. Should use useAppSelector or annotate with RootState
useSelector can't return a primitive value like a string
The component must be wrapped in React.memo to use useSelector

4. Read this slice and identify what cartSlice.actions contains:

const cartSlice = createSlice({
  name: "cart",
  initialState: { items: [], total: 0 },
  reducers: {
    addItem(state, action: PayloadAction<CartItem>) {
      state.items.push(action.payload);
    },
    removeItem(state, action: PayloadAction<string>) {
      state.items = state.items.filter((i) => i.id !== action.payload);
    },
    clearCart(state) {
      state.items = [];
      state.total = 0;
    },
  },
});
{ cart: Function }
{ addItem: string, removeItem: string, clearCart: string }
(x) { addItem: ActionCreator, removeItem: ActionCreator, clearCart: ActionCreator } — three auto-generated action creator functions
{ "cart/addItem": Function, "cart/removeItem": Function, "cart/clearCart": Function }
