Testing Custom Hooks with renderHook

Custom hooks can't be called outside of a component. renderHook solves this — it renders your hook in a minimal wrapper component so you can test it in isolation:

// useCounter.ts — the hook
import { useState } from "react";

const useCounter = (initialValue: number) => {
  const [count, setCount] = useState(initialValue);
  const increment = () => setCount((prev) => prev + 1);
  const decrement = () => setCount((prev) => prev - 1);
  return { count, increment, decrement };
};

export { useCounter };
// useCounter.test.ts — the test
import { renderHook, act } from "@testing-library/react";
import { useCounter } from "./useCounter";

describe("useCounter", () => {
  it("starts with initial value", () => {
    const { result } = renderHook(() => useCounter(5));
    expect(result.current.count).toBe(5);
  });

  it("increments the count", () => {
    const { result } = renderHook(() => useCounter(0));

    act(() => {
      result.current.increment();
    });

    expect(result.current.count).toBe(1);
  });
});
  • renderHook — renders a hook outside of a real component (for isolated testing)
  • result.current — the current return value of the hook (updates after state changes)
  • act() — wraps code that triggers state updates. It tells React: "process this update completely before I make assertions." Required whenever you call functions that change state.

Mocking Modules with vi.mock

"Mocking" means replacing a real module with a fake version you control. This lets you isolate the code you're testing from its dependencies.

Example: you have a component that imports fetchUser from an API module. In tests, you don't want to make real network requests — you replace the module with a fake:

// api/users.ts — the real module (makes network requests)
export const fetchUser = async (id: number): Promise<User> => {
  const response = await fetch(`/api/users/${id}`);
  return response.json();
};
// UserProfile.tsx — component that uses fetchUser
import { useState, useEffect } from "react";
import { fetchUser } from "./api/users";

const UserProfile = ({ userId }: { userId: number }) => {
  const [user, setUser] = useState<{ name: string } | null>(null);

  useEffect(() => {
    fetchUser(userId).then(setUser);
  }, [userId]);

  if (!user) return <p>Loading...</p>;
  return <p>{user.name}</p>;
};
// UserProfile.test.tsx — mock the API module
import { vi } from "vitest";
import { render, screen } from "@testing-library/react";

// Replace "./api/users" with a fake — fetchUser now returns { id: 1, name: "Alice" } instantly
vi.mock("./api/users", () => ({
  fetchUser: vi.fn().mockResolvedValue({ id: 1, name: "Alice" }),
}));

it("displays user name", async () => {
  render(<UserProfile userId={1} />);
  expect(await screen.findByText("Alice")).toBeInTheDocument();
});
  • Without the mock, the test would fail — there's no real server running during tests
  • The fake returns instantly with predictable data — tests are fast and don't depend on external services
  • The mock is applied to the entire test file

vi.fn() — Creating Mock Functions

A "mock function" is a fake function that tracks how it was called (how many times, with what arguments). You use it when a component accepts a callback prop — you pass the mock and then check if the component called it correctly:

// DeleteButton.tsx — component that calls onDelete when clicked
type DeleteButtonProps = {
  itemId: number;
  onDelete: (id: number) => void;
};

const DeleteButton = ({ itemId, onDelete }: DeleteButtonProps) => (
  <button onClick={() => onDelete(itemId)}>Delete</button>
);
// DeleteButton.test.tsx
it("calls onDelete with the item id when clicked", () => {
  const mockOnDelete = vi.fn(); // create a fake function that tracks calls
  render(<DeleteButton itemId={42} onDelete={mockOnDelete} />);

  fireEvent.click(screen.getByRole("button"));

  // Check that our fake was called correctly:
  expect(mockOnDelete).toHaveBeenCalledTimes(1);  // called exactly once
  expect(mockOnDelete).toHaveBeenCalledWith(42);  // called with the right argument
});

vi.spyOn — Check if a Method Was Called

vi.spyOn lets you watch an existing method and check if it was called — without replacing it entirely. Think of it as putting a camera on a function:

// ScrollToTopButton.tsx
const ScrollToTopButton = () => (
  <button onClick={() => window.scrollTo(0, 0)}>Back to top</button>
);
// ScrollToTopButton.test.tsx
it("scrolls to top when clicked", () => {
  // Watch window.scrollTo — track if it gets called
  // .mockImplementation(() => {}) replaces the real behavior with "do nothing"
  // (because jsdom doesn't have real scrolling — without this, it would throw an error)
  const spy = vi.spyOn(window, "scrollTo");
  spy.mockImplementation(() => {});

  render(<ScrollToTopButton />);
  fireEvent.click(screen.getByRole("button"));

  // Was scrollTo called with (0, 0)?
  expect(spy).toHaveBeenCalledWith(0, 0);

  spy.mockRestore(); // remove the spy after test
});

vi.fn() vs vi.spyOn:

  • vi.fn() — creates a brand new fake function (for callback props you pass to components)
  • vi.spyOn(object, "method") — watches a method that already exists on an object (window.scrollTo, console.log, etc.)

Mocking Timers

Some components use setTimeout — for example, showing a notification that disappears after 3 seconds. You don't want your test to actually wait 3 seconds. Fake timers let you fast-forward time instantly:

// Notification.tsx — shows a message, hides after 3 seconds
import { useState, useEffect } from "react";

const Notification = ({ message }: { message: string }) => {
  const [visible, setVisible] = useState(true);

  useEffect(() => {
    const timer = setTimeout(() => setVisible(false), 3000);
    return () => clearTimeout(timer);
  }, []);

  if (!visible) return null; // returning null means "render nothing"
  return <p>{message}</p>;
};
// Notification.test.tsx
import { vi } from "vitest";
import { render, screen, act } from "@testing-library/react";

describe("Notification", () => {
  beforeEach(() => vi.useFakeTimers());  // replace real timers with fakes
  afterEach(() => vi.useRealTimers());   // always restore after test!

  it("disappears after 3 seconds", () => {
    render(<Notification message="Saved!" />);

    // Initially visible
    expect(screen.getByText("Saved!")).toBeInTheDocument();

    // Fast-forward 3 seconds (instant — no real waiting)
    act(() => { vi.advanceTimersByTime(3000); });

    // Now it's gone
    expect(screen.queryByText("Saved!")).not.toBeInTheDocument();
  });
});
  • vi.useFakeTimers() — replaces setTimeout/setInterval with controllable fakes
  • vi.advanceTimersByTime(ms) — fast-forward time by the specified milliseconds (instant, no real waiting)
  • vi.useRealTimers() — restore real timers (always do this in afterEach to avoid affecting other tests)

Mocking react-router

When your component uses routing hooks, you need to mock them:

vi.mock("react-router-dom", async () => {
  const actual = await vi.importActual("react-router-dom");
  return {
    ...actual,                          // keep all real exports
    useNavigate: () => vi.fn(),         // replace only what you need
    useParams: () => ({ id: "42" }),
  };
});

vi.importActual imports the real module so you can keep most of it intact and only override specific exports.
