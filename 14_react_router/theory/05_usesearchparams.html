useSearchParams

An alternative to keeping data in the URL can be search params, for example:

https://online-shop/product/?id=1&name=phone

Let's add one more component to the navigation:

/** App.tsx */

import { BrowserRouter, Route, Routes } from 'react-router-dom';
import ProfilePage from './pages/ProfilePage';
import HomePage from './pages/HomePage';
import ProductPage from './pages/ProductPage';
import TaskPage from './pages/TaskPage';

const App = () => {

  return (
    <BrowserRouter> 
      <Routes> 
        <Route path='home' element={<HomePage/>}/>
        <Route path='profile' element={<ProfilePage/>}/> 
        <Route path='product/:id' element={<ProductPage/>}/>
        <Route path='task' element={<TaskPage/>}/>
      </Routes>
    </BrowserRouter>
  )
}

export default App;

TaskPage implementation:

/** /pages/TaskPage.tsx */

import { useSearchParams } from "react-router-dom";

const TaskPage = () => {

  const [searchParams] = useSearchParams(); 

  return (
    <div>
      <p>Task id: {searchParams.get('id')}</p>
      <p>Task title: {searchParams.get('title')}</p>
    </div>
  )
}

export default TaskPage;

When navigating to

http://localhost:3000/task?id=1&title=work

the page will display something like:

Task id: 1

Task title: work

The useSearchParams hook returns an array where the first element is the searchParams object, which using the get function returns the value of the parameter you specify:

searchParams.get('id') - 1

searchParams.get('title') - work
