Example with returning a result

There may be a case when the request result needs to be received in the component. For this, you simply need to return it from the Thunk:

export const sendUserData = (callback?: () => void) => async (dispatch: AppDispatch, getState: () => RootState) => {
    try {
        const userName = getState().user.name;
        const userSurname = getState().user.surname;

        const response = await fetch('https://yourapi/user', {
            method: 'POST',
            body: JSON.stringify({
                name: userName,
                surname: userSurname,
            })
        });

        const result = await response.json();
        console.log("Result:", result);
        dispatch(setName(''));
        dispatch(setSurname(''));
        if (callback) {
            callback();
        }
        return result; 
    } catch(e) {
        console.log('sendUserData error', e)
    }
};

And handle it in the component:

import { useSelector, useDispatch } from 'react-redux';
import { setName, setSurname, sendUserData } from '../redux/actions';
import { RootState, AppDispatch } from '../redux/store';

const UserData = () => {
  const dispatch = useDispatch<AppDispatch>();

  const name = useSelector((state: RootState) => state.user.name);
  const surname = useSelector((state: RootState) => state.user.surname);

  const successMessage = () => {
    alert('Data sent successfully.');
  }

  return (
    <div>
      <input value={name} onChange={(event) => dispatch(setName(event.target.value))} />
      <input value={surname} onChange={(event) => dispatch(setSurname(event.target.value))} />
      <button onClick={async () => {
        const result = await dispatch(sendUserData(successMessage));
        console.log('Result from View/UI:', result);
      }}>Send</button> 
    </div>
  );
}

export default UserData;

The onClick takes several lines, leaving it in this form is bad practice. It's better to extract it into a separate function:

import { useSelector, useDispatch } from 'react-redux';
import { setName, setSurname, sendUserData } from '../redux/actions';
import { RootState, AppDispatch } from '../redux/store';

const UserData = () => {
  const dispatch = useDispatch<AppDispatch>();

  const name = useSelector((state: RootState) => state.user.name);
  const surname = useSelector((state: RootState) => state.user.surname);

  const successMessage = () => {
    alert('Data sent successfully.');
  }

  const sendData = async () => {
    const result = await dispatch(sendUserData(successMessage));
    console.log('Result from View/UI:', result);
  }

  return (
    <div>
      <input value={name} onChange={(event) => dispatch(setName(event.target.value))} />
      <input value={surname} onChange={(event) => dispatch(setSurname(event.target.value))} />
      <button onClick={sendData}>Send</button> 
    </div>
  );
}

export default UserData;
