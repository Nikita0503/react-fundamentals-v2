Example with a callback parameter

There are cases when you need to pass parameters from the component into a Thunk. For our case: let's pass a function that notifies the user about successful data submission:

/** /components/UserData.js */
import { useSelector, useDispatch } from 'react-redux';
import { setName, setSurname, sendUserData } from '../redux/actions';

const UserData = () => {
  const dispatch = useDispatch();

  const name = useSelector(state => state.user.name);
  const surname = useSelector(state => state.login.surname);

  const successMessage = () => {
    alert('Data sent successfully.');
  }

  return (
    <div>
      <input value={name} onChange={(event) => dispatch(setName(event.target.value))} />
      <input value={surname} onChange={(event) => dispatch(setSurname(event.target.value))} />
      <button onClick={() => {dispatch(sendUserData(successMessage))}}>Send</button> 
    </div>
  );
}

export default UserData;

For this, we create a successMessage function that displays a message and pass it as a parameter to the sendUserData Thunk.

All that remains is to call this function inside the Thunk:

export const sendUserData = (callback) => async (dispatch, getState) => {    
    try{
        const userName = getState().user.name;
        const userSurname = getState().user.surname;

	const response = await fetch('https://yourapi/user', {
            method: 'POST',
            body: JSON.stringify({
                name: userName,
                surname: userSurname,
            })
        });

        const result = await response.json();
        console.log("Result:", result);
	dispatch(setName(''));
        dispatch(setSurname(''));
	if(!!callback){
	    callback();
	}
    }catch(e){
        console.log('sendUserData error', e)
    }
};

This mechanism is called a callback, so let the parameter have that name.

After clearing the name and surname fields, we check whether a callback was passed and call it.

Now after successful request execution, the user will be notified.
